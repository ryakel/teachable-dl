# Floors are set to the versions this fork is tested against.
#
# These are deliberately current rather than permissive: several closed upstream
# issues ("ChromeDriver only supports Chrome version 112", "Not working in
# Chrome 115", "Not Working in Chrome 119") were nothing but a stale selenium or
# seleniumbase, and an out-of-date yt-dlp breaks video extraction whenever the
# host changes its player.
#
# Requires Python 3.10 or newer.

selenium>=4.47.0
seleniumbase>=4.52.2

# Keep this one current above all others; `pip install -U yt-dlp` is the first
# thing to try if videos suddenly stop downloading.
yt-dlp>=2026.8.19

requests>=2.34.2
pyotp>=2.10.0

# Not optional in practice. Teachable's video host serves AES-128 encrypted
# HLS, and without a crypto backend yt-dlp cannot decrypt it itself, so it
# hands the whole stream to ffmpeg -- which downloads sequentially and
# silently discards the parallel-fragment setting. Installing this keeps the
# native downloader, and with it the speed.
pycryptodomex>=3.20.0
